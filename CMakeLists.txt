cmake_minimum_required(VERSION 3.8)
project(dji_serial_bridge)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# ── Find dependencies ──────────────────────────────────────────────────────────
find_package(ament_cmake        REQUIRED)
find_package(rclcpp             REQUIRED)
find_package(std_msgs           REQUIRED)
find_package(geometry_msgs      REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(yaml-cpp REQUIRED)

# ── Generate custom messages ───────────────────────────────────────────────────
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/RobotPose.msg"
  "msg/RefSysStatus.msg"
  "msg/CVTarget.msg"
  "msg/PanelDetection.msg"
  "msg/PanelDetectionArray.msg"
  "msg/TargetState.msg"
  "msg/Ping.msg"
  DEPENDENCIES std_msgs geometry_msgs
)

# Make generated headers available before the executable target is defined.
ament_export_dependencies(rosidl_default_runtime)

# ── Bridge node executable ─────────────────────────────────────────────────────
add_executable(dji_serial_bridge_node
  src/dji_serial_bridge_node.cpp
  src/crc_dji.cpp
)

target_include_directories(dji_serial_bridge_node
  PRIVATE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/rosidl_generator_cpp>
)

ament_target_dependencies(dji_serial_bridge_node
  rclcpp
  std_msgs
  geometry_msgs
)

# Link the generated message type-support to the executable.
rosidl_get_typesupport_target(cpp_typesupport_target
  ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(dji_serial_bridge_node "${cpp_typesupport_target}")

add_executable(serial_debug src/serial_debug.cpp)
target_include_directories(serial_debug PRIVATE include)
add_executable(test_bridge src/test_bridge.cpp)
target_include_directories(test_bridge PRIVATE include)
ament_target_dependencies(test_bridge rclcpp)
target_link_libraries(test_bridge "${cpp_typesupport_target}" yaml-cpp)

# ── Install ────────────────────────────────────────────────────────────────────
install(TARGETS dji_serial_bridge_node serial_debug test_bridge
  DESTINATION lib/${PROJECT_NAME}
)

install(DIRECTORY include/
  DESTINATION include
)

install(DIRECTORY config launch
  DESTINATION share/${PROJECT_NAME}
)

# ── Tests ──────────────────────────────────────────────────────────
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_diagnostic_frames test/test_diagnostic_frames.cpp)
  target_include_directories(test_diagnostic_frames PRIVATE include)
  target_link_libraries(test_diagnostic_frames util)
  # crc_dji.* is generated by Taproot (GPL-3, "DO NOT EDIT"): keep every linter off it.
  set(AMENT_LINT_AUTO_FILE_EXCLUDE
    src/crc_dji.cpp
    include/dji_serial_bridge/crc_dji.hpp
  )
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()


ament_package()
